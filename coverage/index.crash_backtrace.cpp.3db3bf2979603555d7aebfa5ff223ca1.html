52 static void print_backtrace(const CONTEXT* crash_context, const char* reason)
53 {
54 // An abort() from inside the exception filter (or a second crashing thread) would print again.
55 if (InterlockedExchange(&handling_crash, 1))
56 return;
57
58 HANDLE out = INVALID_HANDLE_VALUE;
59 if (backtrace_file_path[0])
60 out = CreateFileA(backtrace_file_path, GENERIC_WRITE, 0, nullptr, CREATE_ALWAYS, FILE_ATTRIBUTE_NORMAL, nullptr);
61 if (out == INVALID_HANDLE_VALUE)
62 out = GetStdHandle(STD_ERROR_HANDLE);
63
64 char line[2048];
65 snprintf(line, sizeof(line), "crashed: %s\n", reason);
66 write_str(out, line);
67
68 HANDLE process = GetCurrentProcess();
69 HANDLE thread = GetCurrentThread();
70 pSymSetOptions(SYMOPT_UNDNAME | SYMOPT_DEFERRED_LOADS | SYMOPT_LOAD_LINES);
71 if (!pSymInitializeW(process, symbol_search_path[0] ? symbol_search_path : nullptr, TRUE))
72 write_str(out, "(could not load symbols)\n");
73
74 // StackWalk64 updates the context as it unwinds.
75 CONTEXT context = *crash_context;
76 STACKFRAME64 frame = {};
77 DWORD machine;
78 #if defined(_M_X64)
79 machine = IMAGE_FILE_MACHINE_AMD64;
80 frame.AddrPC.Offset = context.Rip;
81 frame.AddrFrame.Offset = context.Rbp;
82 frame.AddrStack.Offset = context.Rsp;
83 #elif defined(_M_ARM64)
84 machine = IMAGE_FILE_MACHINE_ARM64;
85 frame.AddrPC.Offset = context.Pc;
86 frame.AddrFrame.Offset = context.Fp;
87 frame.AddrStack.Offset = context.Sp;
88 #elif defined(_M_IX86)
89 machine = IMAGE_FILE_MACHINE_I386;
90 frame.AddrPC.Offset = context.Eip;
91 frame.AddrFrame.Offset = context.Ebp;
92 frame.AddrStack.Offset = context.Esp;
93 #else
94 #error "unknown architecture"
95 #endif
96 frame.AddrPC.Mode = AddrModeFlat;
97 frame.AddrFrame.Mode = AddrModeFlat;
98 frame.AddrStack.Mode = AddrModeFlat;
99
100 for (int i = 0; i < 64; i++)
101 {
102 if (!pStackWalk64(machine, process, thread, &frame, &context, nullptr, pSymFunctionTableAccess64, pSymGetModuleBase64, nullptr))
103 break;
104
105 DWORD64 address = frame.AddrPC.Offset;
106 if (!address)
107 break;
108
109 // Every frame but the first is a return address, which points past the call.
110 DWORD64 lookup_address = i == 0 ? address : address - 1;
111
112 char module_name[MAX_PATH] = "?";
113 DWORD64 module_base = pSymGetModuleBase64(process, address);
114 if (module_base && GetModuleFileNameA((HMODULE)module_base, module_name, sizeof(module_name)))
115 {
116 if (const char* slash = strrchr(module_name, '\\'))
117 memmove(module_name, slash + 1, strlen(slash + 1) + 1);
118 }
119
120 alignas(SYMBOL_INFO) char symbol_buffer[sizeof(SYMBOL_INFO) + 512] = {};
121 SYMBOL_INFO* symbol = (SYMBOL_INFO*)symbol_buffer;
122 symbol->SizeOfStruct = sizeof(SYMBOL_INFO);
123 symbol->MaxNameLen = 512;
124 DWORD64 displacement = 0;
125
126 IMAGEHLP_LINE64 source = {};
127 source.SizeOfStruct = sizeof(source);
128 DWORD line_displacement = 0;
129
130 if (pSymFromAddr(process, lookup_address, &displacement, symbol))
131 {
132 if (pSymGetLineFromAddr64(process, lookup_address, &line_displacement, &source))
133 snprintf(line, sizeof(line), "#%-2d %s!%s [%s:%lu]\n", i, module_name, symbol->Name, source.FileName, source.LineNumber);
134 else
135 snprintf(line, sizeof(line), "#%-2d %s!%s+0x%llx\n", i, module_name, symbol->Name, (unsigned long long)displacement);
136 }
137 else
138 {
139 snprintf(line, sizeof(line), "#%-2d %s+0x%llx\n", i, module_name, (unsigned long long)(address - module_base));
140 }
141 write_str(out, line);
142 }
143
144 if (out != GetStdHandle(STD_ERROR_HANDLE))
145 CloseHandle(out);
146
147 // The C runtime drops whatever is still buffered when the process dies this way, which on
148 // CI (where stdout is a pipe, so fully buffered) is usually all of the program's output.
149 fflush(nullptr);
150 }

152 static LONG WINAPI on_unhandled_exception(EXCEPTION_POINTERS* info)
153 {
154 char reason[128];
155 snprintf(reason, sizeof(reason), "unhandled exception 0x%08lX at %p",
156 info->ExceptionRecord->ExceptionCode, info->ExceptionRecord->ExceptionAddress);
157 print_backtrace(info->ContextRecord, reason);

159 // Let the C runtime's filter carry on as usual (for a C++ exception, it calls std::terminate).
